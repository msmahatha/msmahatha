# `05` · THE ENGINE

```python
class Developer:

    def __init__(self):
        self.curiosity = "∞"
        self.learning = True
        self.projects = []
        self.failures = []
        self.experiments = []

    def build(self, idea):
        problem = understand(idea)
        architecture = design(problem)
        system = implement(architecture)
        result = test(system)

        if result.failed:
            self.failures.append(result)
            return self.improve(system)

        self.projects.append(system)
        return system

    def improve(self, system):
        return self.build(system.problem)

    def evolve(self):
        while self.learning:
            self.experiment()
            self.learn()
            self.build()
```
